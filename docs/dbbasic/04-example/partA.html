A. 기본: 좌표계·측정

count(*) FILTER (WHERE 조건): 조건에 맞는 행만 세는 것으로 한 쿼리에서 여러 관계를 동시에 비교 가능

💻 레이어별 좌표계 점검

  • 모든 레이어의 좌표계를 확인한 후, 좌표계가 다르다면 EPSG:5186으로 변환하여 좌표계를 통일해 봅시다.

    • 💡 HINT! ALTER TABLE, ST_SRID, ST_Transform

    정답 
    
  /* 생성되어 있는 모든 기하 데이터의 좌표계 번호 출력 */
  SELECT * FROM geometry_columns;    
  /* ALTER TABLE로 hospital 테이블의 구조 변경 */
  ALTER TABLE hospital
  ALTER COLUMN geom TYPE geometry(POINT, 5186) 
  USING ST_TRANSFORM(geom, 5186);
  /* 좌표계 변경되었는지 결과 확인 */
  SELECT ST_SRID(geom) FROM hospital LIMIT 1;
  /* ALTER TABLE로 park 테이블의 구조 변경 */
  ALTER TABLE park
  ALTER COLUMN geom TYPE geometry(POINT, 5186) 
  USING ST_TRANSFORM(geom, 5186);
  /* 좌표계 변경되었는지 결과 확인 */
  SELECT ST_SRID(geom) FROM park LIMIT 1;
  /* ALTER TABLE로 road 테이블의 구조 변경 */
  ALTER TABLE road
  ALTER COLUMN geom TYPE geometry(Multilinestring, 5186) 
  USING ST_TRANSFORM(geom, 5186);
  /* 좌표계 변경되었는지 결과 확인 */
  SELECT ST_SRID(geom) FROM road LIMIT 1;
  /* 생성되어 있는 모든 기하 데이터의 좌표계 번호 재출력 */
  SELECT * FROM geometry_columns;

💻 좌표 읽기

  • 소방서 이름과 X, Y 좌표(m)를 소방서 이름으로 정렬하여 5개만 확인해 봅시다.

    • 💡 HINT! ST_X, ST_Y

    정답 
    
  /* ST_X, ST_Y를 이용해서 X, Y 좌표값 생성 */
  SELECT 서센터명, ST_X(geom) AS X, ST_Y(geom) AS Y
  FROM fire
  /* 서센터명으로 정렬 */
  ORDER BY 서센터명
  /* 5개만 확인 */
  LIMIT 5;

💻 좌표 변환

  • 소방서 좌표를 위도·경도로 바꾸고 소방서 이름, X, Y 좌표를 조회하십시오.

    • 💡 HINT! ST_Transform

    정답 
    
  /* ST_X, ST_Y를 이용해서 X, Y 좌표값 생성 */
  SELECT 서센터명, ST_X(ST_Transform(geom, 4326)) AS lat, ST_Y(ST_Transform(geom, 4326)) AS lon
  FROM fire
  /* 5개만 확인 */
  LIMIT 5;

💻 면적 계산

  • 건물에 대한 면적을 계산하되 건물명이 널 값이 아닌 것 중 면적이 큰 건물 5개를 조회하십시오.

    • 💡 HINT! ST_Area

    정답 
    
  SELECT 건물명, ST_Area(geom) AS area
  FROM building
  /* 건물명이 널이 아닌 것을 설정 */
  WHERE 건물명 IS NOT NULL
  /* 5개만 확인 */
  ORDER BY area DESC
  LIMIT 5;

💻 길이 계산

  • 도로 등급(road_rank)별 길이를 구하고, 구간 수(segment), 길이 합계(km)를 조회하십시오.

    • 💡 HINT! ST_Length, SUM

    정답 
    
  /* 구간 수는 count, 길이를 합하기 위해 sum 사용 */
  SELECT road_rank, count(*) as segment, sum(ST_Length(geom)/1000) as length
  FROM road
  /* 도로 등급별로 조회 */
  GROUP BY road_rank
  /* 도로 등급별로 정렬 */
  ORDER BY road_rank;

💻 면적과 둘레 계산

  • 건물의 둘레가 가장 긴 건물 5개의 이름, 면적, 둘레를 구해 보십시오.

    • 💡 HINT! ST_Area, ST_Perimeter

    정답 
     
  /* 건물별 면적과 둘레 계산
  SELECT 건물명, ST_Area(geom) AS area, ST_Perimeter AS perimeter
  FROM building
  /* 계산된 둘레 값으로 정렬 */
  ORDER BY perimeter DESC
  /* 5개만 조회 */
  LIMIT 5;

⚠️ SetSRID와 Transfrom의 구분

  • 좌표계 변환 대신 라벨만 바꾸면 어떤 일이 생기는지 확인하고, SetSRID와 Tranform의 차이를 꼭 기억합시다!

-- 1) 소방서를 4326으로 변환한 연습용 테이블
CREATE TABLE fire_4326 AS
SELECT 서센터명, ST_Transform(geom, 4326) AS geom FROM fire;

-- 2) 서로 다른 SRID 연산 → 오류 발생을 확인
SELECT ST_Distance(h.geom, f.geom)
FROM hospital AS h, fire_4326 AS f
LIMIT 1;

-- 3) 올바른 해결: 좌표를 실제로 변환
SELECT ST_Distance(h.geom, ST_Transform(f.geom, 5186)) AS dist_m
FROM hospital AS h, fire_4326 AS f
LIMIT 1;

-- 4) 잘못된 해결: 라벨만 5186으로 바꿈 → 거리 값이 엉뚱함
SELECT round(ST_Distance(h.geom, ST_SetSRID(f.geom, 5186))) AS wrong_dist_m
FROM hospital h, fire_4326 f
LIMIT 1;

-- 5) 정리
DROP TABLE fire_4326;
